- [ ] **Step 4：实现规范化命令校验**

在读取真实设备、解密 payload 之后，发布 MQTT 之前调用以下纯逻辑。正确顺序是：**读取设备 -> 解密 -> 规范化/校验 -> 计算状态 -> 发布 -> 落库**。

```python
ACTION_ALIASES = {
    ("light", "turn_on"): "on",
    ("light", "turn_off"): "off",
    ("ac", "turn_on"): "on",
    ("ac", "turn_off"): "off",
    ("door_lock", "open"): "unlock",
    ("door_lock", "close"): "lock",
    ("humidifier", "turn_on"): "on",
    ("humidifier", "turn_off"): "off",
    ("smart_plug", "turn_on"): "on",
    ("smart_plug", "turn_off"): "off",
}


def normalize_and_validate_command(
    device_type: str,
    action: str,
    params: dict[str, Any],
) -> tuple[str, dict[str, Any]]:
    canonical = ACTION_ALIASES.get(
        (device_type, action.lower()), action.lower()
    )
    if canonical not in SUPPORTED_ACTIONS_BY_DEVICE_TYPE.get(device_type, ()):
        raise HTTPException(status_code=400, detail="unsupported_device_action")

    def require_range(key: str, low: float, high: float) -> None:
        if key not in params:
            return
        value = params[key]
        if isinstance(value, bool) or not isinstance(value, (int, float)):
            raise HTTPException(status_code=400, detail=f"invalid_{key}")
        if not low <= float(value) <= high:
            raise HTTPException(status_code=400, detail=f"invalid_{key}")

    if device_type == "light":
        require_range("brightness", 0, 100)
    elif device_type == "ac":
        require_range("temp", 16, 30)
    elif device_type == "curtain":
        require_range("position", 0, 100)
    elif device_type == "humidifier":
        require_range("level", 1, 3)
        require_range("target_humidity", 30, 80)
    return canonical, dict(params)
```

`execute_device_command()` 必须用返回的 canonical action 同时构造 MQTT payload、计算状态和写日志，避免设备动作与本地状态语义分裂。

```python
def ensure_device_constraints(conn: sqlite3.Connection) -> None:
    duplicates = conn.execute(
        "SELECT mqtt_topic, COUNT(*) AS count "
        "FROM devices GROUP BY mqtt_topic HAVING COUNT(*) > 1"
    ).fetchall()
    if duplicates:
        topics = ", ".join(row["mqtt_topic"] for row in duplicates)
        raise RuntimeError(f"duplicate mqtt_topic values must be resolved: {topics}")
    conn.execute(
        "CREATE UNIQUE INDEX IF NOT EXISTS ux_devices_mqtt_topic "
        "ON devices(mqtt_topic)"
    )
```

```python
import threading
from collections.abc import Callable
from typing import Any


class DeviceStateProjection:
    def __init__(self) -> None:
        self._lock = threading.Lock()
        self._states: dict[int, dict[str, Any]] = {}

    def rebuild(
        self,
        loader: Callable[[], dict[int, dict[str, Any]]],
    ) -> None:
        with self._lock:
            states = loader()
            self._states = {
                device_id: dict(state) for device_id, state in states.items()
            }

    def update(self, device_id: int, state: dict[str, Any]) -> None:
        with self._lock:
            self._states[device_id] = dict(state)

    def get(self, device_id: int) -> dict[str, Any] | None:
        with self._lock:
            state = self._states.get(device_id)
            return dict(state) if state is not None else None


device_state_projection = DeviceStateProjection()
```

投影只负责进程内规则判断，可由 SQLite 完整重建，不是新的事实存储。

| 历史重复 topic 阻止启动 | Task 0 先检查并备份，迁移不自动删数据 | 回滚 Task 1，恢复备份，人工去重后重试 |
| 动作规范化影响设备协议 | 为 `turn_on/off`、门锁 `open/close` 写兼容测试 | 回滚 Task 1 |
| 场景中途 MQTT 失败 | 发布前解析完整 payload，错误响应明确已执行数量 | 回滚 Task 2；不得声称已执行动作被撤销 |
| 投影与 SQLite 不一致 | SQLite 为事实源，rebuild 与 update 串行化且每个写入口有测试 | 回滚 Task 2 并重启，从 SQLite 恢复旧行为 |
| MQTT 故障被误当进程故障 | `/api/ready` 只用于诊断，Compose 本轮保持不变 | 回滚 Task 3 |
| 开发客户端无法连接 | 不修改端口、CORS 或 `10.0.2.2` | 回滚相关后端提交，不改前端地址 |

回滚使用 `git revert <task-commit>`，禁止 `git reset --hard`。数据库只新增索引，回滚时执行 `DROP INDEX IF EXISTS ux_devices_mqtt_topic`；恢复数据前必须停止 backend。
